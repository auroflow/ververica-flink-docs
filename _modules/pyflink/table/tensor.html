def _build_normalized_tensor_payload(tensors):
    """Build an ndarray exposing contiguous tensor payload bytes.

    The zero-copy path is intentionally narrow. It targets callers that split one
    dense, C-contiguous buffer into adjacent variable-shaped tensor views,
    for example when row offsets describe slices of one values buffer. These adjacent
    views already form the final Arrow payload, so reuse them instead of copying them
    with ``numpy.concatenate()``.

    Independently allocated tensors and gapped or reordered views fall back to
    ``numpy.concatenate()``.
    """
    import numpy as np

    # Single tensor: reuse its normalized payload directly.
    if len(tensors) == 1:
        return tensors[0]

    first = tensors[0]
    start_address = first.__array_interface__["data"][0]
    end_address = start_address + first.nbytes
    # Common case: concatenate the disjoint tensors.
    for tensor in tensors[1:]:
        address = tensor.__array_interface__["data"][0]
        if address != end_address:
            return np.concatenate(tensors, axis=None)
        end_address += tensor.nbytes

    # Each tensor is C-contiguous, and their byte ranges are adjacent. Reuse them
    # when one C-contiguous root covers the complete payload range.
    root = first
    while isinstance(root.base, np.ndarray):
        root = root.base

    root_address = root.__array_interface__["data"][0]
    start_byte = start_address - root_address
    end_byte = end_address - root_address
    if not root.flags.c_contiguous or start_byte < 0 or end_byte > root.nbytes:
        return np.concatenate(tensors, axis=None)

    byte_view = root.view(np.uint8).reshape(-1)
    return byte_view[start_byte:end_byte].view(first.dtype)



[docs]def variable_tensors_to_arrow(tensors, dtype):
    """Convert variable-shaped NumPy tensors to a PyArrow ExtensionArray.

    Use this helper to construct a variable-shaped ``TensorType`` output
    column in a Table API Arrow UDF or DataFrame API Arrow ``map_batches``
    function. Each non-null tensor must be a dense, non-scalar
    ``numpy.ndarray``. Tensor shapes may differ between rows. Tensor values are
    cast to the requested element dtype and encoded in C-contiguous order.

    Args:
        tensors: An iterable containing ``numpy.ndarray`` values or ``None``.
        dtype: Tensor element dtype accepted by
            :meth:`~pyflink.table.types.DataTypes.TENSOR`. It must match the
            corresponding tensor result type declared by the Arrow UDF.

    Returns:
        A ``pyarrow.ExtensionArray`` for a variable-shaped ``TensorType``
        output column.

    The returned Arrow array may share input memory. This happens for a single
    already-normalized tensor or for multiple adjacent tensor views covered by
    one C-contiguous buffer. Do not modify input tensors while the
    Arrow array is in use.

    Example::

        >>> def transpose_tensors(tensor_column):
        ...     tensors = [
        ...         None if tensor is None else tensor.T
        ...         for tensor in tensor_column.to_pylist()
        ...     ]
        ...     return variable_tensors_to_arrow(tensors, dtype="float32")
    """
    import numpy as np
    import pyarrow as pa
    import pyarrow_hotfix  # noqa # pylint: disable=unused-import

    from pyflink.fn_execution.utils._tensor_arrow_ext import (
        PyFlinkVariableTensorArrowType,
        _check_variable_tensor_shape,
    )

    tensor_type = TensorType(dtype)
    numpy_dtype = np.dtype(tensor_type.dtype).newbyteorder("<")
    storage_type = to_arrow_type(tensor_type)
    max_payload_size = np.iinfo(np.int32).max
    shapes = []
    byte_offsets = [0]
    null_mask = []
    non_null_tensors = []
    all_tensors_normalized = True
    has_null = False

    for tensor in tensors:
        if tensor is None:
            shapes.append([])
            byte_offsets.append(byte_offsets[-1])
            null_mask.append(True)
            has_null = True
            continue
        if isinstance(tensor, np.ma.MaskedArray):
            raise TypeError(
                "Variable tensor values do not support element masks; "
                "use None for a null tensor row.")
        if not isinstance(tensor, np.ndarray):
            raise TypeError(
                "Variable tensor values must be numpy.ndarray or None, got %s."
                % type(tensor).__name__)
        tensor = np.asarray(tensor)
        if tensor.dtype == np.dtype(object):
            raise TypeError("Variable tensor values do not support object dtype.")
        _check_variable_tensor_shape(tensor.shape)

        payload_size = tensor.size * numpy_dtype.itemsize
        total_payload_size = byte_offsets[-1] + payload_size
        if total_payload_size > max_payload_size:
            raise OverflowError(
                "Variable tensor payload exceeds the maximum size supported by "
                "Arrow binary storage (%d bytes)." % max_payload_size)

        shapes.append(list(tensor.shape))
        byte_offsets.append(total_payload_size)
        null_mask.append(False)
        non_null_tensors.append(tensor)
        all_tensors_normalized = (
            all_tensors_normalized
            and tensor.dtype == numpy_dtype
            and tensor.flags.c_contiguous
        )

    if not non_null_tensors:
        payload = np.empty(0, dtype=numpy_dtype)
    elif all_tensors_normalized:
        payload = _build_normalized_tensor_payload(non_null_tensors)
    else:
        payload = np.concatenate(
            non_null_tensors,
            axis=None,
            dtype=numpy_dtype,
            casting="unsafe",
        )

    shape_array = pa.array(shapes, type=storage_type.field("shape").type)
    offset_buffer = np.asarray(
        byte_offsets, dtype=np.dtype(np.int32).newbyteorder("<"))
    data_array = pa.Array.from_buffers(
        storage_type.field("data").type,
        len(shapes),
        [None, pa.py_buffer(offset_buffer), pa.py_buffer(payload)],
    )
    mask = pa.array(null_mask, type=pa.bool_()) if has_null else None
    storage = pa.StructArray.from_arrays(
        [shape_array, data_array], fields=list(storage_type), mask=mask)
    extension_type = PyFlinkVariableTensorArrowType(storage_type, tensor_type.dtype)
    return pa.ExtensionArray.from_storage(extension_type, storage)
